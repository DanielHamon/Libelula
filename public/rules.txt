rules_version = '2';                                                                                                                                                    service cloud.firestore {
    match /databases/{database}/documents {

      match /tokens/{tokenId} {
        allow read: if true;
        allow write: if request.auth != null;
      }

      match /usuarios/{uid} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && request.auth.uid == uid;
      }

      match /libros/{libroId}/{document=**} {
        allow read: if request.auth != null;
      }

      match /progreso/{progresoId} {
        allow read, write: if request.auth != null;
      }

      match /clases/{claseId} {
        allow read: if request.auth != null;
        allow create: if request.auth != null;
        allow update: if request.auth != null && (
          resource.data.docenteId == request.auth.uid ||
          request.resource.data.diff(resource.data).affectedKeys().hasOnly(['estudiantes'])
        );
        allow delete: if request.auth != null &&
          resource.data.docenteId == request.auth.uid;
      }

    }
  }
